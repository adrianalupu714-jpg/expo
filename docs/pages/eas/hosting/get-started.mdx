import { Text, View, Button, StyleSheet } from 'react-native';
import { useState } from 'react';

export default function App() {
  const [teamA, setTeamA] = useState(0);
  const [teamB, setTeamB] = useState(0);

  return (
    <View style={styles.container}>
      
      <View style={styles.team}>
        <Text style={styles.title}>Echipa A</Text>
        <Text style={styles.score}>{teamA}</Text>
        <Button title="+1" onPress={() => setTeamA(teamA + 1)} />
        <Button title="-1" onPress={() => setTeamA(teamA - 1)} />
      </View>

      <View style={styles.team}>
        <Text style={styles.title}>Echipa B</Text>
        <Text style={styles.score}>{teamB}</Text>
        <Button title="+1" onPress={() => setTeamB(teamB + 1)} />
        <Button title="-1" onPress={() => setTeamB(teamB - 1)} />
      </View>

      <Button title="Reset" onPress={() => {
        setTeamA(0);
        setTeamB(0);
      }} />

    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center'
  },
  team: {
    margin: 20,
    alignItems: 'center'
  },
  title: {
    fontSize: 24,
    fontWeight: 'bold'
  },
  score: {
    fontSize: 40,
    margin: 10
  }
});
